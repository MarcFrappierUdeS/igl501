1.a

	[ANY x WHERE x : 1..2 then y := y - x END] y > 0
<=>
	!x . x : 1..2 => [y := y - x] y > 0
<=>
	!x . x : 1..2 => y - x > 0
<=>
	y > 2

Validation de la simplification en ProB

!(y) . (y : INTEGER => (!x . (x : 1..2 => y - x > 0) <=> y > 2))


1.b

	[PRE x > 10 THEN CHOICE x := x-2 OR x := x+2 END END] x > 2
<=>
	x > 10 & ([CHOICE x := x-2 OR x := x+2 END ] x > 2)
<=>
	x > 10 & ([x := x-2] x > 0 & [x := x+2] x > 2)
<=>
	x > 10 & x-2 > 2 & x+2 > 2
<=>
	x > 10 & x > 4 & x > 0
<=>
	x > 10

Validation de la simplification en ProB

!x.(x : INTEGER => ((x > 10 & x-2 > 2 & x+2 > 2) <=> x > 10))

	[SELECT x >= 0 THEN x := x-2 WHEN x <=1 THEN x := x+2 END] x > 4
<=>
	(x >= 0 => [x := x-2] x > 4) & (x <=1 => [x := x+2] x > 4)
<=>
	(x >= 0 => x-2 > 4) & (x <=1 => x+2 > 4)
<=>
	(x >= 0 => x > 6) & (x <=1 => x > 2)
<=>
	(x < 0 or x > 6) & (x > 1 or x > 2)
<=>
	x > 6
